FX.j07 = function(card){
  var caixa = card.querySelector('.mini');
  var cheio = card.querySelector('.cheio');
  var marcos = [].slice.call(card.querySelectorAll('.marco'));
  return aoRolar(caixa, function(){
    var linha = caixa.scrollTop + caixa.clientHeight * .55;
    var trilho = card.querySelector('.trilho');
    var alturaTotal = trilho.offsetHeight;
    cheio.style.height = limitar((linha - trilho.offsetTop) / alturaTotal, 0, 1) * 100 + '%';
    marcos.forEach(function(m){
      m.classList.toggle('aceso', m.offsetTop + 10 < linha);
    });
  });
};

/* ── FERRAMENTAS USADAS POR ESTE EFEITO ─────────────── */
var lerp = function(a,b,t){ return a + (b-a)*t; };
var limitar = function(v,min,max){ return Math.max(min, Math.min(max, v)); };

/* progresso de 0 a 1 de um elemento atravessando a área visível */
function progressoNaTela(el){
  var r = el.getBoundingClientRect();
  var alt = window.innerHeight || 800;
  return limitar((alt - r.top) / (alt + r.height), 0, 1);
}

/* laço de animação que só roda enquanto o elemento está visível */
function laco(card, passo){
  var id = null, vivo = false, respondeu = false;
  function rodar(){ passo(); id = requestAnimationFrame(rodar); }
  function ligar(){ if(!vivo){ vivo = true; rodar(); } }
  function desligar(){ if(vivo){ vivo = false; cancelAnimationFrame(id); } }
  var obs = new IntersectionObserver(function(e){
    respondeu = true;
    e[0].isIntersecting ? ligar() : desligar();
  },{threshold:0});
  obs.observe(card);
  /* se o observador nao responder, comeca assim mesmo */
  var espera = setTimeout(function(){ if(!respondeu) ligar(); }, 1500);
  return function(){ clearTimeout(espera); cancelAnimationFrame(id); obs.disconnect(); };
}
