FX.j02 = function(card){
  var caixa = card.querySelector('.mini');
  var fita = card.querySelector('.fita-h');
  var palco = card.querySelector('.palco-h');
  return aoRolar(caixa, function(p){
    var curso = Math.max(0, fita.scrollWidth - palco.clientWidth);
    fita.style.transform = 'translateX(' + (-p * curso) + 'px)';
  });
};
