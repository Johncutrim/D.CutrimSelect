/* ══════════════════════════════════════════════════════════
   16 · REVELAR PALAVRA POR PALAVRA
   O JavaScript quebra a frase em palavras e cada uma entra com um atraso. Manual seria escrever tudo à mão.
   ══════════════════════════════════════════════════════════ */

<!-- ── HTML ──────────────────────────────────────────── -->
<p class="js-split">Sua frase aqui. Cada palavra entra com um atraso.</p>

/* ── CSS DE APOIO ───────────────────────────────────── */
.js-split{font-family:var(--display);font-size:25px;line-height:1.35;color:#fff;max-width:400px}
.js-split .palavra{display:inline-block;opacity:0;transform:translateY(16px);transition:opacity .5s cubic-bezier(.16,1,.3,1),transform .5s cubic-bezier(.16,1,.3,1)}
.js-split .palavra.dentro{opacity:1;transform:none}

/* ── JAVASCRIPT ─────────────────────────────────────── */
FX.j16 = function(card){
  var el = card.querySelector('.js-split');
  if(!el.dataset.pronto){
    el.innerHTML = el.textContent.trim().split(/\s+/)
      .map(function(p){ return '<span class="palavra">' + p + '</span>'; }).join(' ');
    el.dataset.pronto = '1';
  }
  var palavras = el.querySelectorAll('.palavra');
  palavras.forEach(function(p){ p.classList.remove('dentro'); });
  palavras.forEach(function(p,i){
    setTimeout(function(){ p.classList.add('dentro'); }, i * 70);
  });
};

/* ── COMO LIGAR ─────────────────────────────────────── */
/* Troque "document" pelo elemento que envolve a animação,
   se você tiver mais de uma cópia na mesma página. */
FX.j16(document);
